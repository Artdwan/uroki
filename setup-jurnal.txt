<script>
var tok = "";

function esc(s){
  return String(s == null ? "" : s).replace(/[&<>"]/g, function(c){
    return {"&":"&amp;","<":"&lt;",">":"&gt;","\"":"&quot;"}[c]; });
}
function save(k,v){ try{ localStorage.setItem(k,v); }catch(e){} }
function load(k){ try{ return localStorage.getItem(k) || ""; }catch(e){ return ""; } }
function drop(k){ try{ localStorage.removeItem(k); }catch(e){} }

function pill(code, field, val){
  var cls = val === "да" ? "pill on" : (val === "нет" ? "pill off" : "pill");
  var txt = val === "да" ? "да" : (val === "нет" ? "нет" : "—");
  return '<span class="' + cls + '" data-code="' + esc(code) + '" data-f="' + field +
         '" data-v="' + esc(val) + '">' + txt + "</span>";
}

function draw(rows){
  var html = "";
  for(var i=0;i<rows.length;i++){
    var r = rows[i];
    var sc = (r.score !== "" && r.max !== "") ? (r.score + " из " + r.max) : "";
    html += '<div class="jrow"><div class="nm">' + esc(r.name) +
            ' <span class="meta">· ' + esc(r.code) + "</span></div>" +
            '<div class="cells"><span class="lb">был</span>' + pill(r.code,"present",r.present) +
            '<span class="lb">камера</span>' + pill(r.code,"camera",r.camera) +
            '<span class="lb">дз</span>' + pill(r.code,"hw",r.hw) +
            (sc ? '<span class="sc">' + esc(sc) + "</span>" : "") +
            "</div></div>";
  }
  document.getElementById("rows").innerHTML = html;
  document.getElementById("empty").textContent =
    rows.length ? "" : "В этой группе нет активных учеников.";
}

function loadDay(){
  var g = document.getElementById("grp").value;
  var d = document.getElementById("date").value;
  var e = document.getElementById("serr");
  if(!g || !d){ e.textContent = "Выбери группу и дату."; return; }
  e.className = "err"; e.textContent = "Загружаю…";
  google.script.run
    .withSuccessHandler(function(res){
      if(res && res.ok){
        e.textContent = ""; draw(res.rows);
        document.getElementById("dayTitle").textContent = g + " · " + d;
      } else e.textContent = (res && res.error) || "Не получилось загрузить.";
    })
    .withFailureHandler(function(){ e.textContent = "Сервер не ответил."; })
    .apiDay(tok, g, d);
}

document.getElementById("rows").addEventListener("click", function(ev){
  var el = ev.target;
  if(!el.getAttribute || !el.getAttribute("data-f")) return;
  var v = el.getAttribute("data-v");
  var next = v === "да" ? "нет" : (v === "нет" ? "" : "да");
  el.setAttribute("data-v", next);
  el.className = next === "да" ? "pill on" : (next === "нет" ? "pill off" : "pill");
  el.textContent = next === "да" ? "да" : (next === "нет" ? "нет" : "—");
  var e = document.getElementById("serr");
  google.script.run
    .withSuccessHandler(function(res){
      if(res && res.ok){ e.className = "err ok"; e.textContent = "Сохранено."; }
      else e.textContent = (res && res.error) || "Отметка не сохранилась, нажми ещё раз.";
    })
    .withFailureHandler(function(){ e.textContent = "Отметка не сохранилась, нажми ещё раз."; })
    .apiMark(tok, document.getElementById("date").value, el.getAttribute("data-code"),
             document.getElementById("grp").value, el.getAttribute("data-f"), next,
             document.getElementById("lesson").value);
});

function opened(res){
  tok = res.token; save("art_tt", tok);
  document.getElementById("loginBox").className = "hide";
  document.getElementById("app").className = "";
  var sel = document.getElementById("grp"), gs = res.groups || [];
  sel.innerHTML = gs.map(function(g){
    return '<option value="' + esc(g) + '">' + esc(g) + "</option>"; }).join("");
  var d = new Date(), p = function(n){ return (n<10?"0":"") + n; };
  document.getElementById("date").value =
    d.getFullYear() + "-" + p(d.getMonth()+1) + "-" + p(d.getDate());
  if(gs.length) loadDay();
}

document.getElementById("enter").addEventListener("click", function(){
  var p = document.getElementById("pass").value.trim();
  var e = document.getElementById("lerr");
  if(!p){ e.textContent = "Впиши пароль."; return; }
  e.textContent = "Проверяю…";
  google.script.run
    .withSuccessHandler(function(res){
      if(res && res.ok){ e.textContent = ""; opened(res); }
      else e.textContent = (res && res.error) || "Не получилось войти.";
    })
    .withFailureHandler(function(){ e.textContent = "Сервер не ответил."; })
    .apiTeacherLogin(p);
});
document.getElementById("pass").addEventListener("keydown", function(ev){
  if(ev.key === "Enter") document.getElementById("enter").click();
});
document.getElementById("load").addEventListener("click", loadDay);
document.getElementById("out").addEventListener("click", function(){
  drop("art_tt"); tok = "";
  document.getElementById("app").className = "hide";
  document.getElementById("loginBox").className = "";
  document.getElementById("pass").value = "";
});
