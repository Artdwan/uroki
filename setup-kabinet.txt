<script>
var TOKEN = "<?= token ?>";
var tok = "";

function esc(s){
  return String(s == null ? "" : s).replace(/[&<>"]/g, function(c){
    return {"&":"&amp;","<":"&lt;",">":"&gt;","\"":"&quot;"}[c]; });
}
function mark(v){
  if(v === "да") return '<span class="y">да</span>';
  if(v === "нет") return '<span class="n">нет</span>';
  return '<span class="d">—</span>';
}
function save(k,v){ try{ localStorage.setItem(k,v); }catch(e){} }
function load(k){ try{ return localStorage.getItem(k) || ""; }catch(e){ return ""; } }
function drop(k){ try{ localStorage.removeItem(k); }catch(e){} }

function show(res){
  document.getElementById("loginBox").className = "hide";
  document.getElementById("app").className = "";
  document.getElementById("who").textContent = res.name || "Ученик";
  document.getElementById("grp").textContent =
    (res.group ? "Группа " + res.group + " · " : "") + "код " + res.code;

  var s = res.stats || {}, parts = ["занятий <b>" + (s.lessons || 0) + "</b>"];
  if(s.lessons){
    parts.push("был <b>" + s.present + "</b>");
    parts.push("с камерой <b>" + s.camera + "</b>");
    parts.push("домашек сдано <b>" + s.hw + "</b>");
    if(s.pct != null) parts.push("средний результат <b>" + s.pct + " %</b>");
  }
  document.getElementById("sum").innerHTML = parts.join(" · ");

  var mats = res.materials || [], html = "";
  for(var i=0;i<mats.length;i++){
    var m = mats[i], sub = [];
    if(m.subject) sub.push(esc(m.subject));
    if(m.given) sub.push("выдано " + esc(m.given));
    if(m.due) sub.push("срок " + esc(m.due));
    html += '<div class="mat"><div class="t"><b>' + esc(m.title) + "</b>" +
            (sub.length ? "<div>" + sub.join(" · ") + "</div>" : "") + "</div>" +
            '<a class="btn" target="_blank" rel="noopener" href="' + esc(m.href) +
            '">Открыть</a></div>';
  }
  document.getElementById("mats").innerHTML = html;
  document.getElementById("mnone").textContent = mats.length ? "" : "Пока ничего не выдано.";

  var rep = res.report || [], t = "";
  if(rep.length){
    t = "<tr><th>дата</th><th>занятие</th><th>был</th><th>камера</th><th>дз</th>" +
        "<th>результат</th><th>комментарий</th></tr>";
    for(var j=0;j<rep.length;j++){
      var r = rep[j];
      var sc = (r.score != null && r.max) ? (r.score + " из " + r.max) : "—";
      t += "<tr><td>" + esc(r.date) + "</td><td>" + esc(r.lesson) + "</td><td>" +
           mark(r.present) + "</td><td>" + mark(r.camera) + "</td><td>" + mark(r.hw) +
           "</td><td>" + sc + "</td><td>" + esc(r.note) + "</td></tr>";
    }
  }
  document.getElementById("rep").innerHTML = t;
  document.getElementById("rnone").textContent = rep.length ? "" : "Занятий пока не отмечено.";
}

function me(token, quiet){
  google.script.run
    .withSuccessHandler(function(res){
      if(res && res.ok){ tok = token; save("art_t", token); show(res); }
      else { drop("art_t");
        if(!quiet) document.getElementById("lerr").textContent =
          (res && res.error) || "Не получилось загрузить кабинет."; }
    })
    .withFailureHandler(function(err){
      document.getElementById("lerr").textContent = "Сервер не ответил. Обнови страницу.";
    })
    .apiMe(token);
}

document.getElementById("enter").addEventListener("click", function(){
  var c = document.getElementById("code").value.trim();
  var p = document.getElementById("pass").value.trim();
  var e = document.getElementById("lerr");
  if(!c || !p){ e.textContent = "Впиши код и пароль."; return; }
  e.textContent = "Проверяю…";
  google.script.run
    .withSuccessHandler(function(res){
      if(res && res.ok){ e.textContent = ""; me(res.token); }
      else e.textContent = (res && res.error) || "Не получилось войти.";
    })
    .withFailureHandler(function(){ e.textContent = "Сервер не ответил, попробуй ещё раз."; })
    .apiLogin(c, p);
});
document.getElementById("pass").addEventListener("keydown", function(ev){
  if(ev.key === "Enter") document.getElementById("enter").click();
});
document.getElementById("refresh").addEventListener("click", function(){ if(tok) me(tok); });
document.getElementById("out").addEventListener("click", function(){
  drop("art_t"); tok = "";
  document.getElementById("app").className = "hide";
  document.getElementById("loginBox").className = "";
  document.getElementById("pass").value = "";
});

(function(){
  var t = TOKEN || load("art_t");
  if(t) me(t, true);
})();
